\documentclass{article}

\usepackage{hyperref}
\usepackage{ocgx2}
\usepackage{graphicx}

\newcommand{\ExplainHSpace}{0.6em}
\newcommand{\ExplainRaise}{1.8ex}
\newcounter{ocgex}

\newcommand{\ClickExplain}[2]{%
	\stepcounter{ocgex}%
	\edef\ocgid{ex\arabic{ocgex}}%
	\switchocg{\ocgid}{#1}%
	\makebox[0pt][l]{%
		\hspace{\ExplainHSpace}%
		\raisebox{\ExplainRaise}[0pt][0pt]{%
			\begin{ocg}[listintoolbar=never]{EXPLAIN}{\ocgid}{0}%
				\fbox{\footnotesize #2}%
			\end{ocg}%
		}%
	}%
}

\begin{document}
	
	Click: \ClickExplain{\fbox{CLICK}}{%
		\begin{minipage}{0.45\linewidth}
			Small image + text: \\
			\includegraphics[width=\linewidth]{example-image-a}
		\end{minipage}%
	}
	
\end{document}
